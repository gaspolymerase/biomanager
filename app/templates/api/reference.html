Python

import os, requests

BASE = "{{ base }}"
auth = {"Authorization": f"Bearer {os.environ['BM_TOKEN']}"}

def everything(path, **params):
    url = f"{BASE}{path}"
    while url:
        page = requests.get(url, headers=auth, params=params).json()
        yield from page["data"]
        url, params = page["next"], None

living = list(everything("/mice", alive="true"))
requests.post(f"{BASE}/mice/1043/weights", headers=auth, json={"grams": 24.1}).raise_for_status()
